    70→    70→function formatBytes(bytes: number): string {
    71→    71→  if (bytes < 1024) return `${bytes} B`
    72→    72→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(0)} KB`
    73→    73→  return `${(bytes / 1024 / 1024).toFixed(1)} MB`
    74→    74→}
