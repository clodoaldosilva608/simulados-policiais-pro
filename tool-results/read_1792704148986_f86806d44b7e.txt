   102→   102→function AuthView({
   103→   103→  session,
   104→   104→  modo,
   105→   105→  setModo,
   106→   106→  onDone,
   107→   107→  toast,
   108→   108→}: {
   109→   109→  session: SessionApi
   110→   110→  modo: Modo
   111→   111→  setModo: (m: Modo) => void
   112→   112→  onDone: () => void
   113→   113→  toast: ReturnType<typeof useToast>['toast']
   114→   114→}) {
   115→   115→  const [name, setName] = useState('')
   116→   116→  const [email, setEmail] = useState('')
   117→   117→  const [password, setPassword] = useState('')
   118→   118→  const [busy, setBusy] = useState(false)
   119→   119→  const [error, setError] = useState('')
   120→   120→
   121→   121→  const submit = async () => {
   122→   122→    setError('')
   123→   123→    setBusy(true)
   124→   124→    try {
   125→   125→      if (modo === 'registrar') {
   126→   126→        const res = await session.register({ name: name.trim() || undefined, email, password })
   127→   127→        if (!res.ok) {
   128→   128→          setError(res.message ?? 'Não foi possível criar sua conta.')
   129→   129→          return
   130→   130→        }
   131→   131→        if (res.trial && res.trial.started === false) {
   132→   132→          toast({
   133→   133→            title: 'Conta criada!',
   134→   134→            description:
   135→   135→              'Este aparelho já usou o teste grátis. Assine para desbloquear todas as provas.',
   136→   136→          })
   137→   137→        } else {
   138→   138→          toast({
   139→   139→            title: 'Conta criada!',
   140→   140→            description: `Teste grátis de ${SITE.trialDays} dias liberado neste aparelho. Bons estudos!`,
   141→   141→          })
   142→   142→        }
   143→   143→        onDone()
   144→   144→        return
   145→   145→      }
   146→   146→      const res = await session.login({ email, password })
   147→   147→      if (!res.ok) {
   148→   148→        setError(res.message ?? 'E-mail ou senha incorretos.')
   149→   149→        return
   150→   150→      }
   151→   151→      toast({ title: 'Bem-vindo de volta!', description: 'Acesso vinculado a este aparelho.' })
   152→   152→      onDone()
   153→   153→    } catch {
   154→   154→      setError('Falha de conexão. Verifique sua internet e tente novamente.')
   155→   155→    } finally {
   156→   156→      setBusy(false)
   157→   157→    }
   158→   158→  }
   159→   159→
   160→   160→  return (
   161→   161→    <>
   162→   162→      <h1 className="text-2xl font-black tracking-tight text-slate-900">Acessar minha conta</h1>
   163→   163→      <p className="mt-1.5 text-sm leading-relaxed text-slate-500">
   164→   164→        Para usar o app — inclusive o teste grátis de {SITE.trialDays} dias — é obrigatório ter uma conta.
   165→   165→        Ela fica ativa em <strong>um aparelho por vez</strong>, protegendo seu acesso.
   166→   166→      </p>
   167→   167→
   168→   168→      {/* Avisos de segurança */}
   169→   169→      {session.takenOver && (
   170→   170→        <div className="mt-4 flex items-start gap-2.5 rounded-xl border border-amber-300 bg-amber-50 p-3.5">
   171→   171→          <AlertTriangle className="mt-0.5 h-5 w-5 shrink-0 text-amber-600" />
   172→   172→          <p className="text-xs leading-relaxed text-amber-800">
   173→   173→            {session.message || 'Esta conta está sendo usada em outro aparelho.'} Ao entrar aqui, o outro
   174→   174→            aparelho é desconectado automaticamente.
   175→   175→          </p>
   176→   176→        </div>
   177→   177→      )}
   178→   178→      {session.blocked && (
   179→   179→        <div className="mt-4 flex items-start gap-2.5 rounded-xl border border-red-300 bg-red-50 p-3.5">
   180→   180→          <Ban className="mt-0.5 h-5 w-5 shrink-0 text-red-600" />
   181→   181→          <p className="text-xs leading-relaxed text-red-800">
   182→   182→            {session.message || 'Esta conta não pode ser acessada. Fale com o suporte.'}
   183→   183→          </p>
   184→   184→        </div>
   185→   185→      )}
   186→   186→
   187→   187→      {/* Abas criar conta | entrar */}
   188→   188→      <div className="mt-5 grid grid-cols-2 gap-1 rounded-xl border border-slate-200 bg-slate-100 p-1">
   189→   189→        {(
   190→   190→          [
   191→   191→            ['registrar', 'Criar conta', UserPlus],
   192→   192→            ['entrar', 'Entrar', LogIn],
   193→   193→          ] as [Modo, string, typeof UserPlus][]
   194→   194→        ).map(([key, label, Icon]) => (
   195→   195→          <button
   196→   196→            key={key}
   197→   197→            type="button"
   198→   198→            onClick={() => {
   199→   199→              setModo(key)
   200→   200→              setError('')
   201→   201→            }}
   202→   202→            aria-pressed={modo === key}
   203→   203→            className={`flex h-10 items-center justify-center gap-1.5 rounded-lg text-xs font-bold transition-colors sm:text-sm ${
   204→   204→              modo === key ? 'bg-white text-slate-900 shadow-sm' : 'text-slate-500 hover:text-slate-700'
   205→   205→            }`}
   206→   206→          >
   207→   207→            <Icon className="h-4 w-4" /> {label}
   208→   208→          </button>
   209→   209→        ))}
   210→   210→      </div>
   211→   211→
   212→   212→      <Card className="mt-3">
   213→   213→        <CardContent className="space-y-4 p-5">
   214→   214→          {modo === 'registrar' ? (
   215→   215→            <>
   216→   216→              <div className="flex items-center gap-2">
   217→   217→                <UserPlus className="h-5 w-5 text-amber-500" />
   218→   218→                <h2 className="text-base font-bold text-slate-900">Criar minha conta</h2>
   219→   219→                <Badge className="ml-auto gap-1 bg-amber-100 text-amber-800 hover:bg-amber-100">
   220→   220→                  <Sparkles className="h-3 w-3" /> Teste grátis incluso
   221→   221→                </Badge>
   222→   222→              </div>
   223→   223→
   224→   224→              <div className="space-y-1.5">
   225→   225→                <Label htmlFor="reg-name" className="text-sm font-medium text-slate-700">
   226→   226→                  Seu nome <span className="font-normal text-slate-400">(opcional)</span>
   227→   227→                </Label>
   228→   228→                <Input
   229→   229→                  id="reg-name"
   230→   230→                  autoComplete="name"
   231→   231→                  placeholder="Como devemos te chamar?"
   232→   232→                  value={name}
   233→   233→                  onChange={(e) => setName(e.target.value)}
   234→   234→                  className="h-11"
   235→   235→                />
   236→   236→              </div>
   237→   237→            </>
   238→   238→          ) : (
   239→   239→            <>
   240→   240→              <div className="flex items-center gap-2">
   241→   241→                <LogIn className="h-5 w-5 text-amber-500" />
   242→   242→                <h2 className="text-base font-bold text-slate-900">Entrar na minha conta</h2>
   243→   243→              </div>
   244→   244→            </>
   245→   245→          )}
   246→   246→
   247→   247→          <div className="space-y-1.5">
   248→   248→            <Label htmlFor="auth-email" className="text-sm font-medium text-slate-700">
   249→   249→              E-mail
   250→   250→            </Label>
   251→   251→            <Input
   252→   252→              id="auth-email"
   253→   253→              type="email"
   254→   254→              inputMode="email"
   255→   255→              autoComplete="email"
   256→   256→              placeholder="voce@email.com"
   257→   257→              value={email}
   258→   258→              onChange={(e) => {
   259→   259→                setEmail(e.target.value)
   260→   260→                setError('')
   261→   261→              }}
   262→   262→              className="h-11"
   263→   263→            />
   264→   264→          </div>
   265→   265→
   266→   266→          <div className="space-y-1.5">
   267→   267→            <Label htmlFor="auth-pass" className="text-sm font-medium text-slate-700">
   268→   268→              Senha {modo === 'registrar' && <span className="font-normal text-slate-400">(mínimo 6 caracteres)</span>}
   269→   269→            </Label>
   270→   270→            <Input
   271→   271→              id="auth-pass"
   272→   272→              type="password"
   273→   273→              autoComplete={modo === 'registrar' ? 'new-password' : 'current-password'}
   274→   274→              placeholder="••••••••"
   275→   275→              value={password}
   276→   276→              onChange={(e) => {
   277→   277→                setPassword(e.target.value)
   278→   278→                setError('')
   279→   279→              }}
   280→   280→              onKeyDown={(e) => e.key === 'Enter' && !busy && void submit()}
   281→   281→              className="h-11"
   282→   282→            />
   283→   283→          </div>
   284→   284→
   285→   285→          {error && (
   286→   286→            <p className="rounded-lg border border-red-200 bg-red-50 px-3 py-2 text-xs leading-relaxed text-red-700">
   287→   287→              {error}
   288→   288→            </p>
   289→   289→          )}
   290→   290→
   291→   291→          <Button
   292→   292→            onClick={() => void submit()}
   293→   293→            disabled={busy || !email.trim() || password.length < (modo === 'registrar' ? 6 : 1)}
   294→   294→            className="h-12 w-full bg-slate-900 text-base font-bold text-white hover:bg-slate-800"
   295→   295→          >
   296→   296→            {busy ? (
   297→   297→              <>
   298→   298→                <Loader2 className="mr-2 h-5 w-5 animate-spin" /> {modo === 'registrar' ? 'Criando conta...' : 'Entrando...'}
   299→   299→              </>
   300→   300→            ) : modo === 'registrar' ? (
   301→   301→              <>
   302→   302→                <Sparkles className="mr-2 h-5 w-5" /> Criar conta e iniciar teste grátis
   303→   303→              </>
   304→   304→            ) : (
   305→   305→              <>
   306→   306→                <LogIn className="mr-2 h-5 w-5" /> Entrar neste aparelho
   307→   307→              </>
   308→   308→            )}
   309→   309→          </Button>
   310→   310→
   311→   311→          <p className="flex items-start gap-1.5 text-[11px] leading-relaxed text-slate-400">
   312→   312→            <Smartphone className="mt-0.5 h-3.5 w-3.5 shrink-0" />
   313→   313→            Proteção anti-compartilhamento: sua conta funciona em um aparelho por vez. Ao entrar em um novo
   314→   314→            aparelho, o anterior é desconectado (basta entrar de novo para voltar). Dúvidas? {SITE.supportEmail}.
   315→   315→          </p>
   316→   316→        </CardContent>
   317→   317→      </Card>
   318→   318→
   319→   319→      {modo === 'entrar' && (
   320→   320→        <p className="mt-4 text-center text-xs leading-relaxed text-slate-500">
   321→   321→          Já assinou na Cakto? Entre com sua conta e ative com o <strong>e-mail da compra</strong> na próxima
   322→   322→          tela.
   323→   323→        </p>
   324→   324→      )}
   325→   325→    </>
   326→   326→  )
   327→   327→}

   332→   332→
   333→   333→function AccountView({
   334→   334→  session,
   335→   335→  onBack,
   336→   336→  onDone,
   337→   337→  toast,
   338→   338→  caktoOk,
   339→   339→}: {
   340→   340→  session: SessionApi
   341→   341→  onBack: () => void
   342→   342→  onDone: () => void
   343→   343→  toast: ReturnType<typeof useToast>['toast']
   344→   344→  caktoOk: boolean | null
   345→   345→}) {
   346→   346→  const access = session.access
   347→   347→  const status: 'active' | 'trial' | 'none' =
   348→   348→    access?.status === 'active' ? 'active' : access?.status === 'trial' ? 'trial' : 'none'
   349→   349→
   350→   350→  return (
   351→   351→    <>
   352→   352→      <h1 className="text-2xl font-black tracking-tight text-slate-900">Minha conta</h1>
   353→   353→      <p className="mt-1.5 text-sm leading-relaxed text-slate-500">
   354→   354→        Autenticado como <strong className="break-all">{session.user?.email}</strong>
   355→   355→      </p>
   356→   356→
   357→   357→      {session.blocked && (
   358→   358→        <div className="mt-4 flex items-start gap-2.5 rounded-xl border border-red-300 bg-red-50 p-3.5">
   359→   359→          <Ban className="mt-0.5 h-5 w-5 shrink-0 text-red-600" />
   360→   360→          <p className="text-xs leading-relaxed text-red-800">
   361→   361→            {session.message || 'Esta conta não pode ser acessada. Fale com o suporte.'}
   362→   362→          </p>
   363→   363→        </div>
   364→   364→      )}
   365→   365→
   366→   366→      {/* Status de acesso */}
   367→   367→      {status === 'active' && (
   368→   368→        <Card className="mt-5 border-emerald-200 bg-emerald-50">
   369→   369→          <CardContent className="flex items-center gap-3 p-4">
   370→   370→            <BadgeCheck className="h-6 w-6 shrink-0 text-emerald-600" />
   371→   371→            <div className="min-w-0 flex-1">
   372→   372→              <p className="text-sm font-bold text-emerald-900">Assinatura ativa nesta conta</p>
   373→   373→              <p className="text-xs text-emerald-700">Todas as provas liberadas neste aparelho.</p>
   374→   374→            </div>
   375→   375→            <Button onClick={onDone} className="shrink-0 bg-emerald-600 font-bold text-white hover:bg-emerald-700">
   376→   376→              Entrar
   377→   377→            </Button>
   378→   378→          </CardContent>
   379→   379→        </Card>
   380→   380→      )}
   381→   381→
   382→   382→      {status === 'trial' && (
   383→   383→        <Card className="mt-5 border-emerald-200 bg-emerald-50">
   384→   384→          <CardContent className="flex items-center gap-3 p-4">
   385→   385→            <Clock className="h-6 w-6 shrink-0 text-emerald-600" />
   386→   386→            <div className="min-w-0 flex-1">
   387→   387→              <p className="text-sm font-bold text-emerald-900">
   388→   388→                Teste grátis ativo — {(access?.trialDaysLeft ?? 0)}{' '}
   389→   389→                {(access?.trialDaysLeft ?? 0) === 1 ? 'dia restante' : 'dias restantes'}
   390→   390→              </p>
   391→   391→              <p className="text-xs text-emerald-700">Todas as provas liberadas durante o teste.</p>
   392→   392→            </div>
   393→   393→            <Button onClick={onDone} className="shrink-0 bg-emerald-600 font-bold text-white hover:bg-emerald-700">
   394→   394→              Entrar
   395→   395→            </Button>
   396→   396→          </CardContent>
   397→   397→        </Card>
   398→   398→      )}
   399→   399→
   400→   400→      {status === 'none' && !session.blocked && (
   401→   401→        <Card className="mt-5 border-amber-200 bg-amber-50">
   402→   402→          <CardContent className="flex items-center gap-3 p-4">
   403→   403→            <AlertTriangle className="h-6 w-6 shrink-0 text-amber-600" />
   404→   404→            <div className="min-w-0 flex-1">
   405→   405→              <p className="text-sm font-bold text-amber-900">
   406→   406→                {access?.status === 'canceled' ? 'Assinatura cancelada' : 'Acesso expirado'}
   407→   407→              </p>
   408→   408→              <p className="text-xs leading-relaxed text-amber-800">
   409→   409→                Ative sua assinatura abaixo para continuar estudando neste aparelho.
   410→   410→              </p>
   411→   411→            </div>
   412→   412→          </CardContent>
   413→   413→        </Card>
   414→   414→      )}
   415→   415→
   416→   416→      {/* Teste grátis (quando elegível) */}
   417→   417→      {status === 'none' && !session.blocked && (
   418→   418→        <Card className="mt-4 border-dashed">
   419→   419→          <CardContent className="flex flex-col items-center gap-3 p-5 text-center sm:flex-row sm:text-left">
   420→   420→            <div className="flex h-11 w-11 shrink-0 items-center justify-center rounded-xl bg-amber-100">
   421→   421→              <Sparkles className="h-6 w-6 text-amber-600" />
   422→   422→            </div>
   423→   423→            <div className="min-w-0 flex-1">
   424→   424→              <p className="text-sm font-bold text-slate-900">Quero experimentar</p>
   425→   425→              <p className="mt-0.5 text-xs leading-relaxed text-slate-500">
   426→   426→                {SITE.trialDays} dias grátis, sem cartão de crédito. Um teste por aparelho.
   427→   427→              </p>
   428→   428→            </div>
   429→   429→            <TrialButton session={session} toast={toast} />
   430→   430→          </CardContent>
   431→   431→        </Card>
   432→   432→      )}
   433→   433→
   434→   434→      {/* Ativação de assinatura */}

   493→   493→function ActivateArea({
   494→   494→  session,
   495→   495→  toast,
   496→   496→  caktoOk,
   497→   497→}: {
   498→   498→  session: SessionApi
   499→   499→  toast: ReturnType<typeof useToast>['toast']
   500→   500→  caktoOk: boolean | null
   501→   501→}) {
   502→   502→  const [modo, setModo] = useState<Ativacao>('email')
   503→   503→  const [email, setEmail] = useState('')
   504→   504→  const [code, setCode] = useState('')
   505→   505→  const [checking, setChecking] = useState(false)
   506→   506→  const [error, setError] = useState('')
   507→   507→
   508→   508→  const handleActivate = async () => {
   509→   509→    setError('')
   510→   510→    if (modo === 'email') {
   511→   511→      const cleanEmail = email.trim().toLowerCase()
   512→   512→      if (!EMAIL_RE.test(cleanEmail)) {
   513→   513→        setError('Digite o e-mail que você usou na compra na Cakto.')
   514→   514→        return
   515→   515→      }
   516→   516→      setChecking(true)
   517→   517→      try {
   518→   518→        const res = await session.activate({ email: cleanEmail })
   519→   519→        if (res.valid) {
   520→   520→          toast({ title: 'Acesso ativado!', description: 'Assinatura ativa na sua conta. Bons estudos!' })
   521→   521→        } else {
   522→   522→          setError(res.message ?? 'Não encontramos assinatura ativa para este e-mail.')
   523→   523→        }
   524→   524→      } catch {
   525→   525→        setError('Não foi possível verificar agora. Verifique sua conexão e tente novamente.')
   526→   526→      } finally {
   527→   527→        setChecking(false)
   528→   528→      }
   529→   529→      return
   530→   530→    }
   531→   531→
   532→   532→    const clean = code.trim().toUpperCase()
   533→   533→    if (!clean) {
   534→   534→      setError('Digite o código que você recebeu por e-mail após a compra.')
   535→   535→      return
   536→   536→    }
   537→   537→    setChecking(true)
   538→   538→    try {
   539→   539→      const res = await session.activate({ code: clean })
   540→   540→      if (res.valid) {
   541→   541→        toast({ title: 'Acesso ativado!', description: 'Sua assinatura está ativa. Bons estudos!' })
   542→   542→      } else {
   543→   543→        setError(res.message ?? 'Código inválido. Confira o e-mail de compra.')
   544→   544→      }
   545→   545→    } catch {
   546→   546→      setError('Não foi possível verificar o código agora. Verifique sua conexão e tente novamente.')
   547→   547→    } finally {
   548→   548→      setChecking(false)
   549→   549→    }
   550→   550→  }
   551→   551→
   552→   552→  return (
   553→   553→    <>
   554→   554→      <div className="mt-5 grid grid-cols-2 gap-1 rounded-xl border border-slate-200 bg-slate-100 p-1">
   555→   555→        <button
   556→   556→          type="button"
   557→   557→          onClick={() => {
   558→   558→            setModo('email')
   559→   559→            setError('')
   560→   560→          }}
   561→   561→          aria-pressed={modo === 'email'}
   562→   562→          className={`flex h-10 items-center justify-center gap-1.5 rounded-lg text-xs font-bold transition-colors sm:text-sm ${
   563→   563→            modo === 'email' ? 'bg-white text-slate-900 shadow-sm' : 'text-slate-500 hover:text-slate-700'
   564→   564→          }`}
   565→   565→        >
   566→   566→          <Mail className="h-4 w-4" /> E-mail da compra
   567→   567→        </button>
   568→   568→        <button
   569→   569→          type="button"
   570→   570→          onClick={() => {
   571→   571→            setModo('codigo')
   572→   572→            setError('')
   573→   573→          }}
   574→   574→          aria-pressed={modo === 'codigo'}
   575→   575→          className={`flex h-10 items-center justify-center gap-1.5 rounded-lg text-xs font-bold transition-colors sm:text-sm ${
   576→   576→            modo === 'codigo' ? 'bg-white text-slate-900 shadow-sm' : 'text-slate-500 hover:text-slate-700'
   577→   577→          }`}
   578→   578→        >
   579→   579→          <KeyRound className="h-4 w-4" /> Código de acesso
   580→   580→        </button>
   581→   581→      </div>
   582→   582→
   583→   583→      <Card className="mt-3">
   584→   584→        <CardContent className="space-y-4 p-5">
   585→   585→          {modo === 'email' ? (
   586→   586→            <>
   587→   587→              <div className="flex items-center gap-2">
   588→   588→                <Mail className="h-5 w-5 text-amber-500" />
   589→   589→                <h2 className="text-base font-bold text-slate-900">Ativar com e-mail da compra</h2>
   590→   590→                <Badge className="ml-auto gap-1 bg-emerald-100 text-emerald-800 hover:bg-emerald-100">
   591→   591→                  <BadgeCheck className="h-3 w-3" /> Recomendado
   592→   592→                </Badge>
   593→   593→              </div>
   594→   594→              <div className="space-y-1.5">
   595→   595→                <Label htmlFor="access-email" className="text-sm font-medium text-slate-700">
   596→   596→                  E-mail usado no pagamento (Cakto)
   597→   597→                </Label>
   598→   598→                <Input
   599→   599→                  id="access-email"
   600→   600→                  type="email"
   601→   601→                  inputMode="email"
   602→   602→                  autoComplete="email"
   603→   603→                  placeholder="voce@email.com"
   604→   604→                  value={email}
   605→   605→                  onChange={(e) => {
   606→   606→                    setEmail(e.target.value)
   607→   607→                    setError('')
   608→   608→                  }}
   609→   609→                  onKeyDown={(e) => e.key === 'Enter' && !checking && void handleActivate()}
   610→   610→                  className="h-11"
   611→   611→                />
   612→   612→              </div>
   613→   613→              {error && (
   614→   614→                <p className="rounded-lg border border-red-200 bg-red-50 px-3 py-2 text-xs leading-relaxed text-red-700">
   615→   615→                  {error}
   616→   616→                </p>
   617→   617→              )}

   620→   620→                disabled={checking}
   621→   621→                className="h-12 w-full bg-slate-900 text-base font-bold text-white hover:bg-slate-800"
   622→   622→              >
   623→   623→                {checking ? (
   624→   624→                  <>

   632→   632→              </Button>
