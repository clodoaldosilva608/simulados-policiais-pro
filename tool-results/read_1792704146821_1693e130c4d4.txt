    30→interface AccessScreenProps {
    31→  /** Sessão do aluno (conta autenticada) */
    32→  session: SessionApi
    33→  /** Aba inicial: criar conta (padrão) ou entrar */
    34→  initialMode?: Modo
    35→  onBack: () => void
    36→  /** Acesso liberado → ir para a biblioteca */
    37→  onDone: () => void
    38→}
    39→
    40→type Modo = 'registrar' | 'entrar'
    41→type Ativacao = 'email' | 'codigo'
    42→
    43→const EMAIL_RE = /^[^\s@]+@[^\s@]+\.[^\s@]+$/
    44→
    45→export function AccessScreen({ session, initialMode = 'registrar', onBack, onDone }: AccessScreenProps) {
    46→  const { toast } = useToast()
    47→  const [modo, setModo] = useState<Modo>(initialMode)
    48→  const [checkingCakto, setCheckingCakto] = useState<boolean | null>(null)
    49→
    50→  useEffect(() => {
    51→    fetch('/api/license')
    52→      .then((r) => (r.ok ? r.json() : null))
    53→      .then((d: { cakto?: boolean } | null) => setCheckingCakto(Boolean(d?.cakto)))
    54→      .catch(() => setCheckingCakto(null))
    55→  }, [])
    56→
    57→  return (
    58→    <div className="flex min-h-screen flex-col bg-slate-50">
    59→      <header className="sticky top-0 z-40 border-b border-slate-800 bg-slate-950 text-white">
    60→        <div className="mx-auto flex max-w-6xl items-center gap-3 px-4 py-3.5">
    61→          <Button
    62→            onClick={onBack}
    63→            variant="ghost"
    64→            size="icon"
    65→            aria-label="Voltar"
    66→            className="h-9 w-9 shrink-0 text-slate-300 hover:bg-slate-800 hover:text-white"
    67→          >
    68→            <ArrowLeft className="h-5 w-5" />
    69→          </Button>
    70→          <div className="flex h-9 w-9 items-center justify-center rounded-lg bg-amber-400 text-slate-950">
    71→            <ShieldCheck className="h-5 w-5" />
    72→          </div>
    73→          <p className="text-base font-bold">{SITE.name}</p>
    74→        </div>
    75→      </header>
    76→
    77→      <main className="mx-auto flex w-full max-w-md flex-1 flex-col justify-center px-4 py-8">
    78→        <motion.div initial={{ opacity: 0, y: 12 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.3 }}>
    79→          {session.authed ? (
    80→            <AccountView session={session} onBack={onBack} onDone={onDone} toast={toast} caktoOk={checkingCakto} />
    81→          ) : (
    82→            <AuthView session={session} modo={modo} setModo={setModo} onDone={onDone} toast={toast} />
    83→          )}
    84→
    85→          <button
    86→            type="button"
    87→            onClick={onBack}
    88→            className="mx-auto mt-6 flex items-center gap-1.5 text-sm font-semibold text-slate-500 transition-colors hover:text-slate-800"
    89→          >
    90→            <ArrowLeft className="h-4 w-4" /> Voltar para a página inicial
    91→          </button>
    92→        </motion.div>
    93→      </main>
    94→    </div>
    95→  )
    96→}

   101→
   102→function AuthView({
   103→  session,
   104→  modo,
   105→  setModo,
   106→  onDone,
   107→  toast,
   108→}: {
   109→  session: SessionApi
   110→  modo: Modo
   111→  setModo: (m: Modo) => void
   112→  onDone: () => void
   113→  toast: ReturnType<typeof useToast>['toast']
   114→}) {
   115→  const [name, setName] = useState('')
   116→  const [email, setEmail] = useState('')
   117→  const [password, setPassword] = useState('')
   118→  const [busy, setBusy] = useState(false)
   119→  const [error, setError] = useState('')
   120→
   121→  const submit = async () => {
   122→    setError('')
   123→    setBusy(true)
   124→    try {
   125→      if (modo === 'registrar') {
   126→        const res = await session.register({ name: name.trim() || undefined, email, password })
   127→        if (!res.ok) {
   128→          setError(res.message ?? 'Não foi possível criar sua conta.')
   129→          return
   130→        }
   131→        if (res.trial && res.trial.started === false) {
   132→          toast({
   133→            title: 'Conta criada!',
   134→            description:
   135→              'Este aparelho já usou o teste grátis. Assine para desbloquear todas as provas.',
   136→          })
   137→        } else {
   138→          toast({
   139→            title: 'Conta criada!',
   140→            description: `Teste grátis de ${SITE.trialDays} dias liberado neste aparelho. Bons estudos!`,
   141→          })
   142→        }
   143→        onDone()
   144→        return
   145→      }
   146→      const res = await session.login({ email, password })
   147→      if (!res.ok) {
   148→        setError(res.message ?? 'E-mail ou senha incorretos.')
   149→        return
   150→      }
   151→      toast({ title: 'Bem-vindo de volta!', description: 'Acesso vinculado a este aparelho.' })
   152→      onDone()
   153→    } catch {
   154→      setError('Falha de conexão. Verifique sua internet e tente novamente.')
   155→    } finally {
   156→      setBusy(false)
   157→    }
   158→  }
   159→
   160→  return (
   161→    <>
   162→      <h1 className="text-2xl font-black tracking-tight text-slate-900">Acessar minha conta</h1>
   163→      <p className="mt-1.5 text-sm leading-relaxed text-slate-500">
   164→        Para usar o app — inclusive o teste grátis de {SITE.trialDays} dias — é obrigatório ter uma conta.
   165→        Ela fica ativa em <strong>um aparelho por vez</strong>, protegendo seu acesso.
   166→      </p>
   167→
   168→      {/* Avisos de segurança */}
   169→      {session.takenOver && (
   170→        <div className="mt-4 flex items-start gap-2.5 rounded-xl border border-amber-300 bg-amber-50 p-3.5">
   171→          <AlertTriangle className="mt-0.5 h-5 w-5 shrink-0 text-amber-600" />
   172→          <p className="text-xs leading-relaxed text-amber-800">
   173→            {session.message || 'Esta conta está sendo usada em outro aparelho.'} Ao entrar aqui, o outro
   174→            aparelho é desconectado automaticamente.
   175→          </p>
   176→        </div>
   177→      )}
   178→      {session.blocked && (
   179→        <div className="mt-4 flex items-start gap-2.5 rounded-xl border border-red-300 bg-red-50 p-3.5">
   180→          <Ban className="mt-0.5 h-5 w-5 shrink-0 text-red-600" />
   181→          <p className="text-xs leading-relaxed text-red-800">
   182→            {session.message || 'Esta conta não pode ser acessada. Fale com o suporte.'}
   183→          </p>
   184→        </div>
   185→      )}
   186→
   187→      {/* Abas criar conta | entrar */}
   188→      <div className="mt-5 grid grid-cols-2 gap-1 rounded-xl border border-slate-200 bg-slate-100 p-1">
   189→        {(
   190→          [
   191→            ['registrar', 'Criar conta', UserPlus],
   192→            ['entrar', 'Entrar', LogIn],
   193→          ] as [Modo, string, typeof UserPlus][]
   194→        ).map(([key, label, Icon]) => (
   195→          <button
   196→            key={key}
   197→            type="button"
   198→            onClick={() => {
   199→              setModo(key)
   200→              setError('')
   201→            }}
   202→            aria-pressed={modo === key}
   203→            className={`flex h-10 items-center justify-center gap-1.5 rounded-lg text-xs font-bold transition-colors sm:text-sm ${
   204→              modo === key ? 'bg-white text-slate-900 shadow-sm' : 'text-slate-500 hover:text-slate-700'
   205→            }`}
   206→          >
   207→            <Icon className="h-4 w-4" /> {label}
   208→          </button>
   209→        ))}
   210→      </div>
   211→
   212→      <Card className="mt-3">
   213→        <CardContent className="space-y-4 p-5">
   214→          {modo === 'registrar' ? (
   215→            <>
   216→              <div className="flex items-center gap-2">
   217→                <UserPlus className="h-5 w-5 text-amber-500" />
   218→                <h2 className="text-base font-bold text-slate-900">Criar minha conta</h2>
   219→                <Badge className="ml-auto gap-1 bg-amber-100 text-amber-800 hover:bg-amber-100">
   220→                  <Sparkles className="h-3 w-3" /> Teste grátis incluso
   221→                </Badge>
   222→              </div>
   223→
   224→              <div className="space-y-1.5">
   225→                <Label htmlFor="reg-name" className="text-sm font-medium text-slate-700">
   226→                  Seu nome <span className="font-normal text-slate-400">(opcional)</span>
   227→                </Label>
   228→                <Input
   229→                  id="reg-name"
   230→                  autoComplete="name"
   231→                  placeholder="Como devemos te chamar?"
   232→                  value={name}
   233→                  onChange={(e) => setName(e.target.value)}
   234→                  className="h-11"
   235→                />
   236→              </div>
   237→            </>
   238→          ) : (
   239→            <>
   240→              <div className="flex items-center gap-2">
   241→                <LogIn className="h-5 w-5 text-amber-500" />
   242→                <h2 className="text-base font-bold text-slate-900">Entrar na minha conta</h2>
   243→              </div>
   244→            </>
   245→          )}
   246→
   247→          <div className="space-y-1.5">
   248→            <Label htmlFor="auth-email" className="text-sm font-medium text-slate-700">
   249→              E-mail
   250→            </Label>
   251→            <Input
   252→              id="auth-email"
   253→              type="email"
   254→              inputMode="email"
   255→              autoComplete="email"
   256→              placeholder="voce@email.com"
   257→              value={email}
   258→              onChange={(e) => {
   259→                setEmail(e.target.value)
   260→                setError('')
   261→              }}
   262→              className="h-11"
   263→            />
   264→          </div>
   265→
   266→          <div className="space-y-1.5">
   267→            <Label htmlFor="auth-pass" className="text-sm font-medium text-slate-700">
   268→              Senha {modo === 'registrar' && <span className="font-normal text-slate-400">(mínimo 6 caracteres)</span>}
   269→            </Label>
   270→            <Input
   271→              id="auth-pass"
   272→              type="password"
   273→              autoComplete={modo === 'registrar' ? 'new-password' : 'current-password'}
   274→              placeholder="••••••••"
   275→              value={password}
   276→              onChange={(e) => {
   277→                setPassword(e.target.value)
   278→                setError('')
   279→              }}
   280→              onKeyDown={(e) => e.key === 'Enter' && !busy && void submit()}
   281→              className="h-11"
   282→            />
   283→          </div>
   284→
   285→          {error && (
   286→            <p className="rounded-lg border border-red-200 bg-red-50 px-3 py-2 text-xs leading-relaxed text-red-700">
   287→              {error}
   288→            </p>
   289→          )}
   290→
   291→          <Button
   292→            onClick={() => void submit()}
   293→            disabled={busy || !email.trim() || password.length < (modo === 'registrar' ? 6 : 1)}
   294→            className="h-12 w-full bg-slate-900 text-base font-bold text-white hover:bg-slate-800"
   295→          >
   296→            {busy ? (
   297→              <>
   298→                <Loader2 className="mr-2 h-5 w-5 animate-spin" /> {modo === 'registrar' ? 'Criando conta...' : 'Entrando...'}
   299→              </>
   300→            ) : modo === 'registrar' ? (
   301→              <>
   302→                <Sparkles className="mr-2 h-5 w-5" /> Criar conta e iniciar teste grátis
   303→              </>
   304→            ) : (
   305→              <>
   306→                <LogIn className="mr-2 h-5 w-5" /> Entrar neste aparelho
   307→              </>
   308→            )}
   309→          </Button>
   310→
   311→          <p className="flex items-start gap-1.5 text-[11px] leading-relaxed text-slate-400">
   312→            <Smartphone className="mt-0.5 h-3.5 w-3.5 shrink-0" />
   313→            Proteção anti-compartilhamento: sua conta funciona em um aparelho por vez. Ao entrar em um novo
   314→            aparelho, o anterior é desconectado (basta entrar de novo para voltar). Dúvidas? {SITE.supportEmail}.
   315→          </p>
   316→        </CardContent>
   317→      </Card>
   318→
   319→      {modo === 'entrar' && (
   320→        <p className="mt-4 text-center text-xs leading-relaxed text-slate-500">
   321→          Já assinou na Cakto? Entre com sua conta e ative com o <strong>e-mail da compra</strong> na próxima
   322→          tela.
   323→        </p>
   324→      )}
   325→    </>
   326→  )
   327→}

   332→
   333→function AccountView({
   334→  session,
   335→  onBack,
   336→  onDone,
   337→  toast,
   338→  caktoOk,
   339→}: {
   340→  session: SessionApi
   341→  onBack: () => void
   342→  onDone: () => void
   343→  toast: ReturnType<typeof useToast>['toast']
   344→  caktoOk: boolean | null
   345→}) {
   346→  const access = session.access
   347→  const status: 'active' | 'trial' | 'none' =
   348→    access?.status === 'active' ? 'active' : access?.status === 'trial' ? 'trial' : 'none'
   349→
   350→  return (
   351→    <>
   352→      <h1 className="text-2xl font-black tracking-tight text-slate-900">Minha conta</h1>
   353→      <p className="mt-1.5 text-sm leading-relaxed text-slate-500">
   354→        Autenticado como <strong className="break-all">{session.user?.email}</strong>
   355→      </p>
   356→
   357→      {session.blocked && (
   358→        <div className="mt-4 flex items-start gap-2.5 rounded-xl border border-red-300 bg-red-50 p-3.5">
   359→          <Ban className="mt-0.5 h-5 w-5 shrink-0 text-red-600" />
   360→          <p className="text-xs leading-relaxed text-red-800">
   361→            {session.message || 'Esta conta não pode ser acessada. Fale com o suporte.'}
   362→          </p>
   363→        </div>
   364→      )}
   365→
   366→      {/* Status de acesso */}
   367→      {status === 'active' && (
   368→        <Card className="mt-5 border-emerald-200 bg-emerald-50">
   369→          <CardContent className="flex items-center gap-3 p-4">
   370→            <BadgeCheck className="h-6 w-6 shrink-0 text-emerald-600" />
   371→            <div className="min-w-0 flex-1">
   372→              <p className="text-sm font-bold text-emerald-900">Assinatura ativa nesta conta</p>

   375→            <Button onClick={onDone} className="shrink-0 bg-emerald-600 font-bold text-white hover:bg-emerald-700">
   376→              Entrar
   377→            </Button>
   378→          </CardContent>
   379→        </Card>
   380→      )}
   381→
   382→      {status === 'trial' && (
   383→        <Card className="mt-5 border-emerald-200 bg-emerald-50">
   384→          <CardContent className="flex items-center gap-3 p-4">
   385→            <Clock className="h-6 w-6 shrink-0 text-emerald-600" />
   386→            <div className="min-w-0 flex-1">
   387→              <p className="text-sm font-bold text-emerald-900">
   388→                Teste grátis ativo — {(access?.trialDaysLeft ?? 0)}{' '}
   389→                {(access?.trialDaysLeft ?? 0) === 1 ? 'dia restante' : 'dias restantes'}
   390→              </p>
   391→              <p className="text-xs text-emerald-700">Todas as provas liberadas durante o teste.</p>
   392→            </div>
   393→            <Button onClick={onDone} className="shrink-0 bg-emerald-600 font-bold text-white hover:bg-emerald-700">
   394→              Entrar
   395→            </Button>
   396→          </CardContent>
   397→        </Card>
   398→      )}
   399→
   400→      {status === 'none' && !session.blocked && (
   401→        <Card className="mt-5 border-amber-200 bg-amber-50">
   402→          <CardContent className="flex items-center gap-3 p-4">
   403→            <AlertTriangle className="h-6 w-6 shrink-0 text-amber-600" />
   404→            <div className="min-w-0 flex-1">
   405→              <p className="text-sm font-bold text-amber-900">
   406→                {access?.status === 'canceled' ? 'Assinatura cancelada' : 'Acesso expirado'}
   407→              </p>
   408→              <p className="text-xs leading-relaxed text-amber-800">
   409→                Ative sua assinatura abaixo para continuar estudando neste aparelho.
   410→              </p>
   411→            </div>
   412→          </CardContent>
   413→        </Card>
   414→      )}
   415→
   416→      {/* Teste grátis (quando elegível) */}
   417→      {status === 'none' && !session.blocked && (
   418→        <Card className="mt-4 border-dashed">
   419→          <CardContent className="flex flex-col items-center gap-3 p-5 text-center sm:flex-row sm:text-left">
   420→            <div className="flex h-11 w-11 shrink-0 items-center justify-center rounded-xl bg-amber-100">
   421→              <Sparkles className="h-6 w-6 text-amber-600" />
   422→            </div>
   423→            <div className="min-w-0 flex-1">
   424→              <p className="text-sm font-bold text-slate-900">Quero experimentar</p>
   425→              <p className="mt-0.5 text-xs leading-relaxed text-slate-500">
   426→                {SITE.trialDays} dias grátis, sem cartão de crédito. Um teste por aparelho.

   428→            </div>

   433→
   434→      {/* Ativação de assinatura */}
   435→      {status === 'none' && !session.blocked && <ActivateArea session={session} toast={toast} caktoOk={caktoOk} />}
   436→
   437→      {/* Aparelho + sair */}
   438→      <Card className="mt-4">
   439→        <CardContent className="space-y-3 p-4">
   440→          <div className="flex items-center gap-2.5">
   441→            <Smartphone className="h-5 w-5 shrink-0 text-slate-500" />
   442→            <p className="text-xs leading-relaxed text-slate-500">
   443→              Esta conta está ativa <strong>apenas neste aparelho</strong>. Ao entrar em outro aparelho, este
   444→              será desconectado — e vice-versa.
   445→            </p>
   446→          </div>
   447→          <Button
   448→            onClick={async () => {
   449→              await session.logout()
   450→              onBack()
   451→            }}
   452→            variant="outline"
   453→            className="h-10 w-full border-slate-300 font-bold text-slate-700 hover:bg-slate-100"
   454→          >
   455→            <LogOut className="mr-2 h-4 w-4" /> Sair deste aparelho
   456→          </Button>
   457→        </CardContent>
   458→      </Card>
   459→    </>
   460→  )
   461→}
   462→
   463→function TrialButton({ session, toast }: { session: SessionApi; toast: ReturnType<typeof useToast>['toast'] }) {
   464→  const [busy, setBusy] = useState(false)
   465→  return (
   466→    <Button
   467→      onClick={async () => {
   468→        setBusy(true)
   469→        try {
   470→          const res = await session.startTrial()
   471→          if (res.ok) {
   472→            toast({ title: 'Teste grátis iniciado!', description: `${SITE.trialDays} dias com tudo liberado.` })
   473→          } else {
   474→            toast({ title: 'Teste indisponível', description: res.message ?? 'Tente novamente mais tarde.' })
   475→          }
   476→        } finally {
   477→          setBusy(false)
   478→        }
   479→      }}
   480→      disabled={busy}
   481→      className="w-full shrink-0 bg-amber-400 font-black text-slate-950 hover:bg-amber-300 sm:w-auto"
   482→    >
   483→      {busy ? <Loader2 className="mr-2 h-4 w-4 animate-spin" /> : <Sparkles className="mr-1.5 h-4 w-4" />}
   484→      Iniciar teste grátis
   485→    </Button>
   486→  )
   487→}

   489→/* ------------------------------------------------------------------ */
   490→/* Ativação (e-mail da compra | código) — exige conta autenticada      */
   491→/* ------------------------------------------------------------------ */
   492→
   493→function ActivateArea({
   494→  session,
   495→  toast,
   496→  caktoOk,
   497→}: {
   498→  session: SessionApi
   499→  toast: ReturnType<typeof useToast>['toast']
   500→  caktoOk: boolean | null
   501→}) {
   502→  const [modo, setModo] = useState<Ativacao>('email')
   503→  const [email, setEmail] = useState('')
   504→  const [code, setCode] = useState('')
   505→  const [checking, setChecking] = useState(false)
   506→  const [error, setError] = useState('')
   507→
   508→  const handleActivate = async () => {
   509→    setError('')
   510→    if (modo === 'email') {
   511→      const cleanEmail = email.trim().toLowerCase()
   512→      if (!EMAIL_RE.test(cleanEmail)) {
   513→        setError('Digite o e-mail que você usou na compra na Cakto.')
   514→        return
   515→      }
   516→      setChecking(true)
   517→      try {
   518→        const res = await session.activate({ email: cleanEmail })
   519→        if (res.valid) {
   520→          toast({ title: 'Acesso ativado!', description: 'Assinatura ativa na sua conta. Bons estudos!' })
   521→        } else {
