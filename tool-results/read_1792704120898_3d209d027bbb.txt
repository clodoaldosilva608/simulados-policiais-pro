    87→function formatBytes(bytes: number): string {
    88→  if (bytes < 1024) return `${bytes} B`
    89→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(0)} KB`
    90→  return `${(bytes / 1024 / 1024).toFixed(1)} MB`
    91→}
