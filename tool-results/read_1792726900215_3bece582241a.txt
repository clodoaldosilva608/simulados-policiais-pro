   130→function formatBytes(bytes: number): string {
   131→  if (bytes < 1024) return `${bytes} B`
   132→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(0)} KB`
   133→  return `${(bytes / 1024 / 1024).toFixed(1)} MB`
   134→}

   183→/** Item do menu hambúrguer (ação ou link, ex. suporte por e-mail) */
   184→function MenuItem({
   185→  icon: Icon,
   186→  label,
   187→  hint,
   188→  badge,
   189→  onClick,
   190→  href,
   191→}: {
   192→  icon: typeof UserRound
   193→  label: string
   194→  hint?: string
   195→  /** Contador destacado (ex. questões no caderno de erros) */
   196→  badge?: number
   197→  onClick: () => void
   198→  /** Quando definido, renderiza <a> em vez de botão */
   199→  href?: string
   200→}) {
   201→  const inner = (
   202→    <>
   203→      <div className="flex h-9 w-9 shrink-0 items-center justify-center rounded-lg bg-slate-100 text-slate-600">
   204→        <Icon className="h-5 w-5" />
   205→      </div>
   206→      <div className="min-w-0 flex-1">
   207→        <p className="truncate text-sm font-bold text-slate-800">{label}</p>
   208→        {hint && <p className="truncate text-xs text-slate-500">{hint}</p>}
   209→      </div>
   210→      {badge !== undefined && badge > 0 && (
   211→        <span className="flex h-6 min-w-6 shrink-0 items-center justify-center rounded-full bg-slate-900 px-1.5 text-[11px] font-black text-amber-300">
   212→          {badge > 99 ? '99+' : badge}
   213→        </span>
   214→      )}
   215→      <ChevronRight className="h-4 w-4 shrink-0 text-slate-400" />
   216→    </>
   217→  )
   218→  const cls =
   219→    'flex w-full items-center gap-3 rounded-xl px-3 py-2.5 text-left transition-colors hover:bg-slate-100'
   220→  return href ? (
   221→    <a href={href} onClick={onClick} className={cls}>
   222→      {inner}
   223→    </a>
   224→  ) : (
   225→    <button type="button" onClick={onClick} className={cls}>
   226→      {inner}
   227→    </button>
   228→  )
   229→}

   231→function OfficialPackCard({
   232→  pack,
   233→  saved,
   234→  onOpen,
   235→  index,
   236→}: {
   237→  pack: SimuladoPack
   238→  saved?: SavedInfo
   239→  onOpen: (id: string) => void
   240→  index: number
   241→}) {
   242→  const Icon = CARREIRA_ICONS[pack.carreira ?? ''] ?? ShieldCheck

   244→  const formatLabel = packFormatLabel(pack)
   245→  const isCE = formatLabel === 'Certo/Errado'
   246→  return (
   247→    <motion.div
   248→      initial={{ opacity: 0, y: 10 }}
   249→      animate={{ opacity: 1, y: 0 }}
   250→      transition={{ delay: Math.min(index * 0.05, 0.25) }}
   251→    >
   252→      <Card className="flex h-full flex-col border-slate-200 transition-shadow hover:shadow-lg">
   253→        <CardContent className="flex flex-1 flex-col gap-3 p-5">
   254→          <div className="flex items-start gap-3">
   255→            <div className={`flex h-12 w-12 shrink-0 items-center justify-center rounded-xl ${CARREIRA_STYLES[pack.carreira ?? ''] ?? 'bg-slate-900 text-amber-300'}`}>
   256→              <Icon className="h-6 w-6" />
   257→            </div>
   258→            <div className="min-w-0 flex-1">
   259→              <div className="flex flex-wrap items-center gap-2">
   260→                <h3 className="text-base font-bold leading-tight text-slate-900">{pack.title}</h3>
   261→                {pack.oficial ? (
   262→                  <Badge className="gap-1 bg-emerald-100 text-emerald-800 hover:bg-emerald-100">
   263→                    <BadgeCheck className="h-3 w-3" /> Prova oficial
   264→                  </Badge>
   265→                ) : (
   266→                  <Badge className="gap-1 bg-amber-100 text-amber-800 hover:bg-amber-100">
   267→                    <PenTool className="h-3 w-3" /> Inédito no estilo da banca
   268→                  </Badge>
   269→                )}
   270→              </div>
   271→              <p className="mt-0.5 text-sm text-slate-500">{pack.subtitle}</p>
   272→            </div>
   273→          </div>
   274→
   275→          <div className="flex flex-wrap items-center gap-2">
   276→            <Badge
   277→              variant="secondary"
   278→              className={
   279→                isCE
   280→                  ? 'bg-emerald-100 text-emerald-800 hover:bg-emerald-100'
   281→                  : 'bg-slate-100 text-slate-800 hover:bg-slate-100'
   282→              }
   283→            >
   284→              {isCE ? <CheckCheck className="mr-1 h-3.5 w-3.5" /> : <ListOrdered className="mr-1 h-3.5 w-3.5" />}
   285→              {formatLabel}
   286→            </Badge>
   287→            <Badge variant="secondary" className="bg-slate-100 text-slate-700 hover:bg-slate-100">
   288→              {pack.ano} • {pack.banca}
   289→            </Badge>
   290→            <Badge variant="secondary" className="bg-slate-100 text-slate-700 hover:bg-slate-100">
   291→              {pack.questions.length} questões
   292→            </Badge>
   293→            <Badge variant="secondary" className="bg-slate-100 text-slate-700 hover:bg-slate-100">
   294→              {subjectsCount} disciplinas
   295→            </Badge>
   296→            <Badge variant="secondary" className="bg-slate-100 text-slate-700 hover:bg-slate-100">
   297→              {formatDurationLabel(pack.durationMinutes)}
   298→            </Badge>
   299→          </div>
   300→
   301→          <div className="mt-auto flex gap-2 pt-1">
   302→            {saved && !saved.finished && (
   303→              <Button onClick={() => onOpen(pack.id)} className="flex-1 bg-emerald-600 font-bold text-white hover:bg-emerald-700">
   304→                <Play className="mr-1.5 h-4 w-4" /> Continuar
   305→              </Button>
   306→            )}
   307→            <Button
   308→              onClick={() => onOpen(pack.id)}
   309→              className={`${saved && !saved.finished ? '' : 'flex-1'} bg-slate-900 font-bold text-white hover:bg-slate-800`}
   310→            >
   311→              {saved ? 'Abrir' : (
   312→                <>
   313→                  <Play className="mr-1.5 h-4 w-4" /> Iniciar
   314→                </>
   315→              )}
   316→            </Button>
   317→          </div>
   318→        </CardContent>
   319→      </Card>
   320→    </motion.div>
   321→  )
   322→}
