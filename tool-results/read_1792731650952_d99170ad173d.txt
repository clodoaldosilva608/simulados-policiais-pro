     1→'use client'
     2→
     3→import { useCallback, useEffect, useRef, useState } from 'react'
     4→import {
     5→  AlertTriangle,
     6→  CheckCircle2,
     7→  FileText,
     8→  Loader2,
     9→  Sparkles,
    10→  Upload,
    11→  X,
    12→} from 'lucide-react'
    13→import { Button } from '@/components/ui/button'
    14→import {
    15→  Dialog,
    16→  DialogContent,
    17→  DialogDescription,
    18→  DialogHeader,
    19→  DialogTitle,
    20→} from '@/components/ui/dialog'
    21→import { Input } from '@/components/ui/input'
    22→import { Label } from '@/components/ui/label'
    23→import { Progress } from '@/components/ui/progress'
    24→import { useToast } from '@/hooks/use-toast'
    25→import { extractPdfText, chunkText } from '@/lib/pdf-text'
    26→import { openPdfForRender, renderPdfPageToJpegDataUrl } from '@/lib/pdf-ocr'
    27→import {
    28→  putLocalDocument,
    29→  getLocalDocument,
    30→  type LocalDocument,
    31→  type LocalDocStyle,
    32→} from '@/lib/local-db'
    33→import { questionsPerChunk, type GeneratedQuestion } from '@/lib/question-gen'
    34→import type { Question, TextBlock } from '@/data/types'
    35→
    36→interface UploadDialogProps {
    37→  open: boolean
    38→  onOpenChange: (open: boolean) => void
    39→  /** Documento em processamento a ser retomado (id) */
    40→  resumeDocId: string | null
    41→  /** Arquivo pré-selecionado via drag & drop */
    42→  initialFile: File | null
    43→  onCompleted: () => void
    44→}
    45→
    46→type Step = 'form' | 'extracting' | 'ocr' | 'generating' | 'done' | 'error'
    47→
    48→const MAX_FILE_SIZE = 50 * 1024 * 1024 // 50 MB
    49→
    50→export function UploadDialog({ open, onOpenChange, resumeDocId, initialFile, onCompleted }: UploadDialogProps) {
    51→  const { toast } = useToast()
    52→
    53→  const [step, setStep] = useState<Step>('form')
    54→  const [file, setFile] = useState<File | null>(null)
    55→  const [title, setTitle] = useState('')
    56→  const [style, setStyle] = useState<LocalDocStyle>('multipla')
    57→  const [docId, setDocId] = useState<string | null>(null)
    58→  const [chunksDone, setChunksDone] = useState(0)
    59→  const [chunksTotal, setChunksTotal] = useState(0)
    60→  const [pages, setPages] = useState<number | null>(null)
    61→  const [charCount, setCharCount] = useState<number | null>(null)
    62→  const [ocrPagesDone, setOcrPagesDone] = useState(0)
    63→  const [ocrPagesTotal, setOcrPagesTotal] = useState<number | null>(null)
    64→  const [errorMsg, setErrorMsg] = useState('')
    65→  const cancelled = useRef(false)
    66→  const fileInputRef = useRef<HTMLInputElement>(null)
    67→
    68→  /** Converte questões geradas pela IA no formato do app e acumula no doc */
    69→  const mergeGenerated = useCallback(
    70→    (doc: LocalDocument, chunkIndex: number, generated: GeneratedQuestion[]): LocalDocument => {
    71→      const questions: Question[] = [...doc.questions]
    72→      const textBlocks: Record<string, TextBlock> = { ...doc.textBlocks }
    73→      let created = 0
    74→      for (const g of generated) {
    75→        let textRef: string | undefined
    76→        if (g.textBlock) {
    77→          textRef = `t${chunkIndex + 1}-${created + 1}`
    78→          textBlocks[textRef] = {
    79→            id: textRef,
    80→            title: g.textBlock.title,
    81→            paragraphs: g.textBlock.paragraphs,
    82→            source: g.textBlock.source,
    83→          }
    84→        }
    85→        questions.push({
    86→          id: questions.length + 1,
    87→          subjectId: g.topic,
    88→          textRef,
    89→          statement: g.statement,
    90→          options: g.options.map((o) => ({ key: o.key as Question['options'][number]['key'], html: o.html })),
    91→          answer: g.answer as Question['answer'],
    92→          comment: g.comment,
    93→        })
    94→        created++
    95→      }
    96→      return { ...doc, questions, textBlocks }
    97→    },
    98→    [],
    99→  )
   100→
   101→  /** Loop de geração: 1 chamada por bloco, salvando progresso no banco local */
   102→  const runGeneration = useCallback(
   103→    async (doc: LocalDocument) => {
   104→      setStep('generating')
   105→      setDocId(doc.id)
   106→      setChunksTotal(doc.chunksTotal)
   107→      setChunksDone(doc.chunksDone)
   108→      setPages(doc.pages)
   109→      setCharCount(doc.charCount)
   110→
   111→      let current = { ...doc }
   112→      const style = doc.style
   113→      const count = questionsPerChunk(style)
   114→      const licenseCode =
   115→        typeof window !== 'undefined' ? window.localStorage.getItem('spp-license-v1') : null
   116→      let accessHeader: Record<string, string> = {}
   117→      try {
   118→        if (licenseCode) {
   119→          const parsed = JSON.parse(licenseCode) as { code?: string }
   120→          if (parsed.code) accessHeader = { 'x-access-code': parsed.code }
   121→        }
   122→      } catch {
   123→        /* sem código */
   124→      }
   125→
   126→      while (!cancelled.current && current.chunksDone < current.chunksTotal) {
   127→        const index = current.chunksDone
   128→        const chunk = current.chunks[index]
   129→        try {
   130→          const res = await fetch('/api/generate', {
   131→            method: 'POST',
   132→            headers: { 'Content-Type': 'application/json', ...accessHeader },
   133→            body: JSON.stringify({ chunk, style, count }),
   134→          })
   135→          const data = (await res.json()) as { questions?: GeneratedQuestion[]; error?: string }
   136→          if (res.ok && Array.isArray(data.questions) && data.questions.length > 0) {
   137→            current = mergeGenerated(current, index, data.questions)
   138→          }
   139→          // Bloco com falha é pulado e o fluxo continua
   140→        } catch {
   141→          /* rede falhou: pula bloco */
   142→        }
   143→
   144→        current = {
   145→          ...current,
   146→          chunksDone: index + 1,
   147→          status: index + 1 >= current.chunksTotal ? 'ready' : 'processing',
   148→        }
   149→        setChunksDone(current.chunksDone)
   150→        try {
   151→          await putLocalDocument(current)
   152→        } catch {
   153→          /* segue em memória */
   154→        }
   155→      }
   156→
   157→      if (cancelled.current) return
   158→
   159→      if (current.questions.length === 0) {
   160→        const failed: LocalDocument = {
   161→          ...current,
   162→          status: 'error',
   163→          error:
   164→            'Não foi possível gerar questões com a IA neste momento. Verifique sua conexão e tente novamente em alguns minutos.',
   165→        }
   166→        try {
   167→          await putLocalDocument(failed)
   168→        } catch {
   169→          /* noop */
   170→        }
   171→        setStep('error')
   172→        setErrorMsg(failed.error ?? '')
   173→        return
   174→      }
   175→
   176→      setStep('done')
   177→      onCompleted()
   178→      toast({
   179→        title: 'Simulado pronto!',
   180→        description: `${current.questions.length} questões geradas e salvas neste aparelho.`,
   181→      })
   182→    },
   183→    [mergeGenerated, onCompleted, toast],
   184→  )
   185→
   186→  /** Retoma a geração de um documento salvo como "processing" */
   187→  const resumeGeneration = useCallback(
   188→    async (id: string) => {
   189→      const doc = await getLocalDocument(id)
   190→      if (!doc) {
   191→        setStep('error')
   192→        setErrorMsg('Simulado não encontrado no armazenamento local.')
   193→        return
   194→      }
   195→      if (doc.status === 'ready') {
   196→        setStep('done')
   197→        return
   198→      }
   199→      if (doc.ocrPagesTotal && doc.chunksTotal === 0 && doc.questions.length === 0) {
   200→        // OCR interrompido no meio (o arquivo original não fica salvo no aparelho)
   201→        setStep('error')
   202→        setErrorMsg(
   203→          'A leitura por IA deste PDF ficou incompleta. Envie o arquivo novamente para gerar o simulado do zero.',
   204→        )
   205→        return
   206→      }
   207→      await runGeneration(doc)
   208→    },
   209→    [runGeneration],
   210→  )
   211→
   212→  // Reset ao abrir (adiado para evitar setState síncrono no effect)
   213→  useEffect(() => {
   214→    if (!open) {
   215→      cancelled.current = true
   216→      return
   217→    }
   218→    cancelled.current = false
   219→    const timer = window.setTimeout(() => {
   220→      setErrorMsg('')
   221→      if (resumeDocId) {
   222→        setFile(null)
   223→        setStep('generating')
   224→        void resumeGeneration(resumeDocId)
   225→      } else if (initialFile && initialFile.size > MAX_FILE_SIZE) {
   226→        // Arquivo vindo do drag & drop não passa pelo input — valida aqui também
   227→        setFile(null)
   228→        setStep('error')
   229→        setErrorMsg('Arquivo muito grande (máximo 50 MB).')
   230→      } else {
   231→        setFile(initialFile)
   232→        setTitle(initialFile ? initialFile.name.replace(/\.pdf$/i, '').replace(/[_-]+/g, ' ') : '')
   233→        setStep('form')
   234→        setDocId(null)
   235→        setChunksDone(0)
   236→        setChunksTotal(0)
   237→        setPages(null)
   238→        setCharCount(null)
   239→        setOcrPagesDone(0)
   240→        setOcrPagesTotal(null)
   241→      }
   242→    }, 0)
   243→    return () => window.clearTimeout(timer)
   244→  }, [open, resumeDocId, initialFile, resumeGeneration])
   245→
   246→  /** Header de acesso (código/assinatura) para as chamadas de IA */
   247→  const buildAccessHeader = useCallback((): Record<string, string> => {
   248→    try {
   249→      const raw = window.localStorage.getItem('spp-license-v1')
   250→      if (raw) {
   251→        const parsed = JSON.parse(raw) as { code?: string }
   252→        if (parsed.code) return { 'x-access-code': parsed.code }
   253→      }
   254→    } catch {
   255→      /* sem código */
   256→    }
   257→    return {}
   258→  }, [])
   259→
   260→  /**
   261→   * Leitura por IA (OCR) de PDF em imagem: renderiza as páginas no navegador e
   262→   * envia lotes de 3 para /api/transcribe. Devolve o texto + total de páginas,
   263→   * ou null se cancelado.
   264→   */
   265→  const runOcr = useCallback(
   266→    async (
   267→      buffer: Uint8Array,
   268→      baseDoc: LocalDocument,
   269→      accessHeader: Record<string, string>,
   270→    ): Promise<{ text: string; totalPages: number; unavailable: boolean } | null> => {
   271→      setStep('ocr')
   272→      const pdf = await openPdfForRender(buffer)
   273→      const totalPages = pdf.numPages
   274→      setPages(totalPages)
   275→      setOcrPagesTotal(totalPages)
   276→      setOcrPagesDone(0)
   277→
   278→      const BATCH = 3
   279→      let ocrText = ''
   280→      let unavailable = false
   281→      for (let start = 1; start <= totalPages; start += BATCH) {
   282→        if (cancelled.current) return null
   283→        const end = Math.min(start + BATCH - 1, totalPages)
   284→        const dataUrls: string[] = []
   285→        for (let p = start; p <= end; p++) {
   286→          if (cancelled.current) return null
   287→          dataUrls.push(await renderPdfPageToJpegDataUrl(pdf, p))
   288→        }
   289→        let batchText = ''
   290→        for (let attempt = 0; attempt < 2 && !batchText; attempt++) {
   291→          try {
   292→            const res = await fetch('/api/transcribe', {
   293→              method: 'POST',
   294→              headers: { 'Content-Type': 'application/json', ...accessHeader },
   295→              body: JSON.stringify({ images: dataUrls }),
   296→            })
   297→            const data = (await res.json()) as { text?: string; error?: string }
   298→            if (res.ok && typeof data.text === 'string' && data.text.trim()) batchText = data.text
   299→            else if (res.status === 503 || res.status === 401) unavailable = true
   300→          } catch {
   301→            /* rede falhou: tenta de novo */
   302→          }
   303→        }
   304→        if (batchText) ocrText = ocrText ? `${ocrText}\n\n${batchText}` : batchText
   305→        setOcrPagesDone(end)
   306→        try {
   307→          await putLocalDocument({ ...baseDoc, ocrText, ocrPagesDone: end })
   308→        } catch {
   309→          /* segue em memória */
   310→        }
   311→        if (!batchText && start === 1) break // 1ª leva falhou (ex.: IA indisponível): não insiste nas demais
   312→      }
   313→      return { text: ocrText.trim(), totalPages, unavailable }
   314→    },
   315→    [],
   316→  )
   317→
   318→  const handleGenerate = async () => {
   319→    if (!file) return
   320→    setErrorMsg('')
   321→    setStep('extracting')
   322→    try {
   323→      const baseTitle = (title.trim() || file.name.replace(/\.pdf$/i, '').replace(/[_-]+/g, ' ') || 'Simulado de PDF').slice(0, 120)
   324→      const accessHeader = buildAccessHeader()
   325→
   326→      let text = ''
   327→      let pagesCount: number | null = null
   328→      try {
   329→        // buffer próprio: o pdf.js desanexa o ArrayBuffer após abrir o documento
   330→        const extraction = await extractPdfText(new Uint8Array(await file.arrayBuffer()))
   331→        text = extraction.text
   332→        pagesCount = extraction.pages
   333→      } catch {
   334→        /* PDF sem texto aproveitável (imagem/scan) — cai para a leitura por IA */
   335→      }
   336→
   337→      // 2) Sem texto suficiente → leitura por IA das páginas (OCR)
   338→      if (text.trim().length < 200) {
   339→        const baseDoc: LocalDocument = {
   340→          id: `pdf-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`,
   341→          title: baseTitle,
   342→          fileName: file.name,
   343→          fileSize: file.size,
   344→          pages: 0,
   345→          charCount: 0,
   346→          style,
   347→          status: 'processing',
   348→          error: null,
   349→          chunks: [],
   350→          chunksDone: 0,
   351→          chunksTotal: 0,
   352→          createdAt: new Date().toISOString(),
   353→          questions: [],
   354→          textBlocks: {},
   355→          ocrUsed: true,
   356→          ocrText: '',
   357→          ocrPagesDone: 0,
   358→          ocrPagesTotal: 0,
   359→        }
   360→        try {
   361→          await putLocalDocument(baseDoc)
   362→        } catch {
   363→          /* segue em memória */
   364→        }
   365→        setDocId(baseDoc.id)
   366→        onCompleted()
   367→
   368→        const ocr = await runOcr(new Uint8Array(await file.arrayBuffer()), baseDoc, accessHeader)
   369→        if (cancelled.current || ocr === null) return
   370→        if (!ocr.text || ocr.text.length < 200) {
   371→          const failed: LocalDocument = {
   372→            ...baseDoc,
   373→            status: 'error',
   374→            error: ocr.unavailable
   375→              ? 'O serviço de leitura por IA está temporariamente indisponível. Tente novamente em alguns minutos.'
   376→              : 'Não foi possível ler o conteúdo deste PDF nem com a leitura por IA. Verifique se o arquivo contém texto legível e tente novamente.',
   377→          }
   378→          try {
   379→            await putLocalDocument(failed)
   380→          } catch {
   381→            /* noop */
   382→          }
   383→          setStep('error')
   384→          setErrorMsg(failed.error ?? '')
   385→          return
   386→        }
   387→        text = ocr.text
   388→
   389→        const chunks = chunkText(text)
   390→        const finalDoc: LocalDocument = {
   391→          ...baseDoc,
   392→          pages: ocr.totalPages,
   393→          charCount: text.length,
   394→          ocrText: text,
   395→          ocrPagesDone: ocr.totalPages,
   396→          chunks,
   397→          chunksTotal: chunks.length,
   398→        }
   399→        try {
   400→          await putLocalDocument(finalDoc)
   401→        } catch {
   402→          /* segue em memória */
   403→        }
   404→        setCharCount(text.length)
   405→        setChunksTotal(chunks.length)
   406→        setChunksDone(0)
   407→        await runGeneration(finalDoc)
   408→        return
   409→      }
   410→
   411→      // 3) Fluxo normal — texto extraído com sucesso
   412→      const chunks = chunkText(text)
   413→      if (chunks.length === 0) {
   414→        setStep('error')
   415→        setErrorMsg('O PDF não possui conteúdo de texto aproveitável.')
   416→        return
   417→      }
   418→
   419→      setPages(pagesCount)
   420→      setCharCount(text.length)
   421→      setChunksTotal(chunks.length)
   422→      setChunksDone(0)
   423→
   424→      // 4) Salva o documento no banco LOCAL e inicia a geração
   425→      const doc: LocalDocument = {
   426→        id: `pdf-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`,
   427→        title: baseTitle,
   428→        fileName: file.name,
   429→        fileSize: file.size,
   430→        pages: pagesCount ?? 0,
   431→        charCount: text.length,
   432→        style,
   433→        status: 'processing',
   434→        error: null,
   435→        chunks,
   436→        chunksDone: 0,
   437→        chunksTotal: chunks.length,
   438→        createdAt: new Date().toISOString(),
   439→        questions: [],
   440→        textBlocks: {},
   441→      }
   442→      try {
   443→        await putLocalDocument(doc)
   444→      } catch {
   445→        /* segue em memória mesmo sem storage */
   446→      }
   447→      setDocId(doc.id)
   448→      onCompleted()
   449→      await runGeneration(doc)
   450→    } catch (err) {
   451→      setStep('error')
   452→      setErrorMsg(err instanceof Error ? err.message : 'Erro inesperado ao processar o PDF')
   453→    }
   454→  }
   455→
   456→  const chooseFile = () => {
   457→    fileInputRef.current?.click()
   458→  }
   459→
   460→  const stepIcon = (current: Step, target: Step) => {
   461→    const order: Step[] = ['form', 'extracting', 'ocr', 'generating', 'done']
   462→    const done = order.indexOf(current) > order.indexOf(target) || current === 'done'
   463→    if (done) return <CheckCircle2 className="h-4 w-4 text-emerald-600" />
   464→    if (current === target) return <Loader2 className="h-4 w-4 animate-spin text-amber-600" />
   465→    return <span className="h-4 w-4 rounded-full border-2 border-slate-300" />
   466→  }
   467→
   468→  const busy = step === 'extracting' || step === 'ocr' || step === 'generating'
   469→
   470→  return (
   471→    <Dialog open={open} onOpenChange={onOpenChange}>
   472→      <DialogContent className="sm:max-w-lg">
   473→        <input
   474→          ref={fileInputRef}
   475→          type="file"
   476→          accept="application/pdf,.pdf"
   477→          className="hidden"
   478→          aria-label="Selecionar arquivo PDF"
   479→          onChange={(e) => {
   480→            const f = e.target.files?.[0]
   481→            if (f) {
   482→              if (f.size > MAX_FILE_SIZE) {
   483→                setStep('error')
   484→                setErrorMsg('Arquivo muito grande (máximo 50 MB).')
   485→                e.target.value = ''
   486→                return
   487→              }
   488→              setFile(f)
   489→              setTitle(f.name.replace(/\.pdf$/i, '').replace(/[_-]+/g, ' '))
   490→              setStep('form')
   491→            }
   492→            e.target.value = ''
   493→          }}
   494→        />
   495→        <DialogHeader>
   496→          <DialogTitle className="flex items-center gap-2">
   497→            <Sparkles className="h-5 w-5 text-amber-500" />
   498→            {resumeDocId ? 'Processando simulado' : 'Transformar PDF em simulado'}
   499→          </DialogTitle>
   500→          <DialogDescription>
   501→            {resumeDocId
   502→              ? 'A geração de questões foi retomada.'
   503→              : 'A IA lê o conteúdo do PDF e cria questões inéditas com gabarito comentado. O arquivo fica salvo apenas no seu aparelho.'}
   504→          </DialogDescription>
   505→        </DialogHeader>
   506→
   507→        {step === 'form' && (
   508→          <div className="space-y-5">
   509→            {/* Zona de arquivo */}
   510→            {file ? (
   511→              <div className="flex items-center gap-3 rounded-lg border border-emerald-200 bg-emerald-50 px-4 py-3">
   512→                <FileText className="h-8 w-8 shrink-0 text-emerald-600" />
   513→                <div className="min-w-0 flex-1">
   514→                  <p className="truncate text-sm font-semibold text-slate-900">{file.name}</p>
   515→                  <p className="text-xs text-slate-500">{(file.size / 1024 / 1024).toFixed(2)} MB</p>
   516→                </div>
   517→                <Button variant="ghost" size="icon" onClick={() => setFile(null)} aria-label="Remover arquivo">
   518→                  <X className="h-4 w-4" />
   519→                </Button>
   520→              </div>
   521→            ) : (
   522→              <button
   523→                type="button"
   524→                onClick={chooseFile}
   525→                className="flex w-full flex-col items-center gap-2 rounded-xl border-2 border-dashed border-slate-300 bg-slate-50 px-6 py-8 text-center transition-colors hover:border-amber-400 hover:bg-amber-50/50"
   526→              >
   527→                <Upload className="h-8 w-8 text-slate-400" />
   528→                <span className="text-sm font-semibold text-slate-700">Clique para escolher um PDF</span>
   529→                <span className="text-xs text-slate-400">Editais, apostilas, resumos • até 50 MB</span>
   530→              </button>
   531→            )}
   532→
   533→            <div className="space-y-1.5">
   534→              <Label htmlFor="doc-title" className="text-sm font-medium text-slate-700">
   535→                Título do simulado
   536→              </Label>
   537→              <Input
   538→                id="doc-title"
   539→                placeholder="Ex.: Direito Administrativo — Capítulo 3"
   540→                value={title}
   541→                onChange={(e) => setTitle(e.target.value)}
   542→              />
   543→            </div>
   544→
   545→            <div>
   546→              <p className="mb-2 text-sm font-medium text-slate-700">Estilo das questões</p>
   547→              <div className="grid gap-2 sm:grid-cols-2">
   548→                <button
   549→                  type="button"
   550→                  onClick={() => setStyle('multipla')}
   551→                  aria-pressed={style === 'multipla'}
   552→                  className={`rounded-xl border-2 p-3 text-left transition-all ${
   553→                    style === 'multipla'
   554→                      ? 'border-amber-500 bg-amber-50 ring-2 ring-amber-400/40'
   555→                      : 'border-slate-200 bg-white hover:border-slate-300'
   556→                  }`}
   557→                >
   558→                  <p className="text-sm font-semibold text-slate-900">Múltipla escolha</p>
   559→                  <p className="mt-0.5 text-xs text-slate-500">5 alternativas (a–e), uma correta</p>
   560→                </button>

   577→            <Button
   578→              onClick={() => void handleGenerate()}

   582→              <Sparkles className="mr-2 h-5 w-5" /> Gerar simulado

   585→        )}
   586→
   587→        {(step === 'extracting' || step === 'ocr' || step === 'generating' || step === 'done') && (

   598→                    </span>

   608→                      {' '}

   612→                </span>

   625→              <li className="flex items-center gap-2.5">
   626→                {stepIcon(step, 'done')}
   627→                <span className={step === 'done' ? 'font-semibold text-slate-900' : 'text-slate-600'}>

   629→                </span>
   630→              </li>
   631→            </ul>
   632→
   633→            {step === 'ocr' && ocrPagesTotal !== null && ocrPagesTotal > 0 && (

   649→
   650→            {chunksTotal > 0 && (

   652→                <div className="mb-1.5 flex justify-between text-xs text-slate-500">

   662→                </p>
   663→              </div>
   664→            )}
   665→
   666→            {step === 'done' && (
   667→              <Button
   668→                onClick={() => onOpenChange(false)}
   669→                className="h-11 w-full bg-emerald-600 font-bold text-white hover:bg-emerald-700"

   674→          </div>

   676→
   677→        {step === 'error' && (
   678→          <div className="space-y-4 py-1">
   679→            <div className="flex items-start gap-3 rounded-lg border border-red-200 bg-red-50 px-4 py-3">

   687→              {docId && (

   689→                  onClick={() => void resumeGeneration(docId)}

   691→                >
   692→                  Tentar novamente

   695→              <Button variant="outline" onClick={() => onOpenChange(false)} className="flex-1 border-slate-300">
   696→                Fechar
   697→              </Button>
   698→            </div>
   699→          </div>

   701→      </DialogContent>
   702→    </Dialog>
